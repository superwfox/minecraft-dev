<template>
  <section v-if="tree" class="project-tree" :class="`pt-${mode}`" aria-label="项目结构树">
    <header class="pt-header">
      <div><span class="pt-eyebrow">项目结构</span><h2>从模块到每一个文件</h2></div>
      <div class="pt-header-meta"><span>{{ tree.modules.length }} 个模块</span><span>{{ tree.nodes.length }} 个文件</span><span>修订 {{ tree.revision }}</span></div>
    </header>
    <div class="pt-scroll" tabindex="0" aria-label="可滚动的项目结构图">
      <div class="pt-map">
        <div v-lattice:one class="pt-root">
          <div class="pt-root-kicker"><Blocks :size="17" aria-hidden="true"/> 插件项目</div>
          <h3 data-font="dynamic">{{ tree.project.projectName || '正在规划项目' }}</h3>
          <code data-font="dynamic">{{ tree.project.packageName || '待确定包名' }}</code>
          <p data-font="dynamic">{{ tree.project.summary || '结构树会随规划逐步展开' }}</p>
          <div class="pt-platform"><span>{{ genTask.coreType }} {{ genTask.version }}</span><span>Java {{ tree.project.javaVersion }}</span></div>
          <div v-if="mode === 'progress'" class="pt-root-progress">
            <div><strong>{{ totals.done }}<small> / {{ totals.total }}</small></strong><span>文件已完成</span></div>
            <div class="pt-meter"><span :style="{ width: `${totals.total ? totals.done / totals.total * 100 : 0}%` }"/></div>
            <span>契约 {{ totals.passed }} / {{ totals.contracts }} 项</span>
          </div>
          <span v-else class="pt-root-note">{{ mode === 'draft' ? '正在构建结构树' : '待你确认后开始生成' }}</span>
        </div>
        <div class="pt-modules">
          <div v-for="module in tree.modules" :key="module.id" class="pt-branch">
            <button v-lattice:card type="button" class="pt-module" :class="{ selected: genTask.treeFocusIds.includes(module.id), 'pt-system': module.kind === 'system' }"
                    :aria-pressed="mode === 'confirm' ? genTask.treeFocusIds.includes(module.id) : undefined" @click="openTarget(module.id)">
              <span class="pt-module-top"><Folder :size="15" aria-hidden="true"/><span>{{ module.kind === 'system' ? '系统模块' : module.kind === 'common' ? '公共模块' : '业务模块' }}</span><Check v-if="mode === 'confirm' && genTask.treeFocusIds.includes(module.id)" :size="14" aria-hidden="true"/></span>
              <strong data-font="dynamic">{{ module.title }}</strong>
              <code data-font="dynamic">{{ moduleSuffix(module.id) }}</code>
              <p data-font="dynamic">{{ module.responsibility }}</p>
              <span v-if="mode === 'progress'" class="pt-module-count">{{ progress[module.id].done }} / {{ progress[module.id].total }} 已完成</span>
              <span v-else class="pt-module-count">{{ progress[module.id].total }} 个文件</span>
            </button>
            <div class="pt-nodes">
              <button v-for="node in tree.nodes.filter(node => node.moduleId === module.id)" :key="node.id" v-lattice:card type="button"
                      class="pt-node" :class="[status(node).status, { selected: genTask.treeFocusIds.includes(node.id) }]"
                      :aria-pressed="mode === 'confirm' ? genTask.treeFocusIds.includes(node.id) : undefined" :aria-busy="mode === 'progress' && status(node).status === 'generating'" @click="openTarget(node.id)">
                <span class="pt-node-icon"><FileCode2 v-if="node.fqcn" :size="19" aria-hidden="true"/><FileText v-else :size="19" aria-hidden="true"/></span>
                <span class="pt-node-body"><span class="pt-node-name"><strong data-font="dynamic">{{ node.id }}</strong><span v-if="mode === 'confirm' && genTask.treeFocusIds.includes(node.id)" class="pt-selected-mark"><Check :size="13" aria-hidden="true"/></span></span>
                  <span class="pt-node-subtitle"><span data-font="dynamic">{{ node.title }}</span><span class="pt-kind">{{ generatorLabels[node.kind] }}</span></span>
                  <span v-if="node.tag === 'gui' || node.origin === 'review' || status(node).buildError" class="pt-badges"><span v-if="node.tag === 'gui'">GUI</span><span v-if="node.origin === 'review'">审查追加</span><span v-if="status(node).buildError" class="pt-error-badge">编译错误</span></span>
                </span>
                <span class="pt-node-tail"><span v-if="mode === 'progress'" class="pt-state-label"><span class="pt-status-dot"/>{{ stateLabels[status(node).status] }}</span><span class="pt-contract-count">{{ contractLabel(node) }}</span><ChevronRight :size="14" aria-hidden="true"/></span>
              </button>

            </div>
          </div>
        </div>
      </div>
    </div>
    <footer class="pt-footer"><span><MousePointer2 :size="13" aria-hidden="true"/>{{ mode === 'confirm' ? '点选模块或文件，查看契约并指定修正范围' : '点击文件查看职责、契约与源码' }}</span><span v-if="mode === 'progress'" class="pt-legend"><i/> 待生成 <i class="busy"/> 生成中 <i class="complete"/> 已完成</span><span v-else>模块对应子包 · 节点对应文件</span></footer>
    <details v-if="mode === 'draft'" class="pt-operations" :open="genTask.treeOpsLog.length < 4">
      <summary><span><GitBranch :size="14" aria-hidden="true"/>规划操作记录</span><span>本轮 {{ genTask.treeToolCallCount }} 次工具调用 · 第 {{ genTask.treeRound || 1 }} 轮</span></summary>
      <ol><li v-for="(operation, index) in genTask.treeOpsLog.slice(-20)" :key="index" data-font="dynamic">{{ operation }}</li><li v-if="!genTask.treeOpsLog.length">正在准备结构规划…</li></ol>
    </details>
    <ProjectTreeDrawer v-if="drawerTarget" :target-id="drawerTarget" @close="drawerTarget = ''" @navigate="drawerTarget = $event"/>
  </section>

<script setup lang="ts">
import { computed, ref } from "vue";
import { Blocks, Check, ChevronRight, FileCode2, FileText, Folder, GitBranch, MousePointer2 } from "lucide-vue-next";
import { genTask } from "../logic/generateState";
import { generatorLabels, moduleProgress, nodeStatusOf, treeTotals, type TreeNodeView } from "../logic/projectTree";
import ProjectTreeDrawer from "./ProjectTreeDrawer.vue";
const props = defineProps<{ mode: "draft" | "confirm" | "progress" }>();
const tree = computed(() => genTask.tree);
const drawerTarget = ref("");
const totals = computed(() => tree.value ? treeTotals(tree.value, genTask.files, genTask.nodeChecks) : { done: 0, total: 0, passed: 0, contracts: 0 });
const progress = computed(() => tree.value ? moduleProgress(tree.value, genTask.files) : {});
const stateLabels = { pending: "待生成", generating: "生成中", done: "已完成", error: "待修复" };
function status(node: TreeNodeView) { return nodeStatusOf(node, genTask.files, genTask.nodeChecks, genTask.buildDiagnostics); }
function moduleSuffix(id: string) { return id === "_root" ? "根包 / Main.java" : id === "_build" ? "pom.xml" : id === "_resources" ? "src/main/resources" : `.${id}`; }
function contractLabel(node: TreeNodeView) {
  const check = genTask.nodeChecks[node.id];
  if (check) return check.total ? `契约 ${check.passed}/${check.total}` : "无契约";
  return node.id === "pom.xml" ? "无契约" : "契约待核对";
}
function openTarget(id: string) {
  if (props.mode === "confirm") {
    const index = genTask.treeFocusIds.indexOf(id);
    if (index >= 0) genTask.treeFocusIds.splice(index, 1); else genTask.treeFocusIds.push(id);
  }
  drawerTarget.value = id;
}
</script>
